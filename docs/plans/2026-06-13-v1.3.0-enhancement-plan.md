```python
# core/log_rotator.py
import os
import gzip
import shutil
from datetime import datetime
from pathlib import Path

class LogRotator:
    MAX_LOG_SIZE = 5 * 1024 * 1024  # 5MB
    MAX_LOG_FILES = 5
    LOG_DIR = 'logs'
    
    def __init__(self, log_file='process_priority_log.txt'):
        self.log_file = log_file
        self.log_path = Path(LOG_DIR) / log_file
        self._ensure_dir()
    
    def _ensure_dir(self):
        self.log_path.parent.mkdir(parents=True, exist_ok=True)
    
    def should_rotate(self) -> bool:
        if not self.log_path.exists():
            return False
        return self.log_path.stat().st_size >= self.MAX_LOG_SIZE
    
    def rotate(self):
        """日志轮转：压缩旧日志，创建新日志"""
        if not self.should_rotate():
            return
        
        timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
        rotated_name = f"{self.log_path.stem}_{timestamp}.txt.gz"
        rotated_path = self.log_path.parent / rotated_name
        
        # 压缩旧日志
        with open(self.log_path, 'rb') as f_in:
            with gzip.open(rotated_path, 'wb') as f_out:
                shutil.copyfileobj(f_in, f_out)
        
        # 清空当前日志
        self.log_path.unlink()
        self.log_path.touch()
        
        # 清理过期日志
        self._clean_old_logs()
    
    def _clean_old_logs(self):
        """删除最旧的日志文件，保持 MAX_LOG_FILES 个"""
        logs = sorted(self.log_path.parent.glob(f"{self.log_path.stem}_*.txt.gz"))
        for old_log in logs[:-self.MAX_LOG_FILES]:
            old_log.unlink()
    
    def get_log_size(self) -> int:
        return self.log_path.stat().st_size if self.log_path.exists() else 0
```

```python
        self.blacklist: List[str] = []

    def _load(self):
        if os.path.exists(self.ORIGINAL_PRIORITY_FILE):
            with open(self.ORIGINAL_PRIORITY_FILE, 'r', encoding='utf-8') as f:
                data = json.load(f)

        with open(self.ORIGINAL_PRIORITY_FILE, 'w', encoding='utf-8') as f:
            json.dump({
                'priorities': self.original_priorities,
                'blacklist': self.blacklist,
                'saved_at': datetime.now().isoformat()
            }, f, indent=2, ensure_ascii=False)
    
    def record_original_priority(self, process_name: str, priority: int):
        """记录进程原始优先级"""
        if process_name not in self.original_priorities:
            self.original_priorities[process_name] = priority
            self.save()
    
    def add_to_blacklist(self, process_name: str):
        """添加到黑名单"""
        if process_name.lower() not in [x.lower() for x in self.blacklist]:
            self.blacklist.append(process_name)
            self.save()
    
    def remove_from_blacklist(self, process_name: str):
        """从黑名单移除"""
```

```python
# ml/rl_agent.py 扩展
class RLPriorityAgent:
    # 状态空间: [cpu_percent, memory_percent, category_type, hour_of_day, is_gaming]
    STATE_DIM = 5
    
    # 动作空间: 调整权重 +/- 5%
    ACTION_SPACE = [-0.05, -0.02, 0, 0.02, 0.05]
    
    def __init__(self):
        self.q_table = {}  # 简化为Q表
        self.learning_rate = 0.1
        self.discount_factor = 0.9
        self.epsilon = 0.1
    
    def get_state(self, cpu: float, memory: float, category: str, hour: int, is_gaming: bool) -> tuple:
        return (
            int(cpu / 20),  # 5个区间
            int(memory / 20),
            hash(category) % 10,
            hour,
            1 if is_gaming else 0
        )
    
    def choose_action(self, state: tuple) -> int:
        if state not in self.q_table:
            self.q_table[state] = [0.0] * len(self.ACTION_SPACE)
        
        if np.random.random() < self.epsilon:
            return np.random.randint(len(self.ACTION_SPACE))
        
        return np.argmax(self.q_table[state])
    
    def learn(self, state, action, reward, next_state):
        if state not in self.q_table:
            self.q_table[state] = [0.0] * len(self.ACTION_SPACE)
        if next_state not in self.q_table:
            self.q_table[next_state] = [0.0] * len(self.ACTION_SPACE)
        
        current_q = self.q_table[state][action]
        max_next_q = max(self.q_table[next_state])
        new_q = current_q + self.learning_rate * (reward + self.discount_factor * max_next_q - current_q)
        self.q_table[state][action] = new_q
    
    def calculate_reward(self, process: dict, system_metrics: dict) -> float:
        """计算奖励：用户满意度 + 系统健康度"""
        cpu = system_metrics.get('cpu_percent', 0)
        memory = system_metrics.get('memory_percent', 0)
        
        # 游戏进程获得高优先级 = 正奖励
        if process.get('category') == 'gaming' and process.get('priority') in ['high', 'above_normal']:
            reward = 10.0
        # 系统高负载时降低非关键进程 = 正奖励
        elif cpu > 80 and process.get('priority') in ['below_normal', 'idle']:
            reward = 5.0
        # 正常情况
        else:
            reward = 0.0
        
        # 惩罚
        if cpu > 90 and process.get('category') not in ['system', 'security']:
            reward -= 5.0
        
        return reward
```

```python
class RLTrainer:
    def __init__(self, agent: RLPriorityAgent, history_manager):
        self.agent = agent
        self.history_manager = history_manager
    
    def train_from_history(self, episodes: int = 100):
        """从历史数据训练"""
        for episode in range(episodes):
            snapshots = self.history_manager.get_recent_snapshots(minutes=60)
            
            for i in range(len(snapshots) - 1):
                current = snapshots[i]
                next_snapshot = snapshots[i + 1]
                
                state = self.agent.get_state(
                    current['cpu_percent'],
                    current['memory_percent'],
                    current['category'],
                    datetime.now().hour,
                    current['category'] == 'gaming'
                )
                
                action = self.agent.choose_action(state)
                
                system_metrics = {'cpu_percent': next_snapshot['cpu_percent']}
                reward = self.agent.calculate_reward(current, system_metrics)
                
                next_state = self.agent.get_state(
                    next_snapshot['cpu_percent'],
                    next_snapshot['memory_percent'],
                    next_snapshot['category'],
                    datetime.now().hour,
                    next_snapshot['category'] == 'gaming'
                )
                
                self.agent.learn(state, action, reward, next_state)
            
            if episode % 10 == 0:
                print(f"Episode {episode}/{episodes} completed")
        
        self.save_model()

        with open('ml/models/rl_agent.pkl', 'wb') as f:
```

```python
import threading
from typing import Callable, Dict, Optional

try:
    import keyboard
    KEYBOARD_AVAILABLE = True
except ImportError:
    KEYBOARD_AVAILABLE = False

class GlobalShortcutManager:
    def __init__(self):
        self.registered_shortcuts: Dict[str, Callable] = {}
        self.listen_thread: Optional[threading.Thread] = None
        self.running = False
    
    def register(self, hotkey: str, callback: Callable, description: str = ""):
        """注册快捷键"""
        if not KEYBOARD_AVAILABLE:
            print("keyboard库不可用，快捷键功能禁用")
            return False
        
        try:
            keyboard.add_hotkey(hotkey, callback)
            self.registered_shortcuts[hotkey] = {
                'callback': callback,
                'description': description
            }
            print(f"快捷键已注册: {hotkey} - {description}")
            return True
        except Exception as e:
            print(f"注册快捷键失败 {hotkey}: {e}")
            return False
    
    def unregister(self, hotkey: str):
        """注销快捷键"""
        if hotkey in self.registered_shortcuts:
            keyboard.remove_hotkey(hotkey)
            del self.registered_shortcuts[hotkey]
    
    def unregister_all(self):
        """注销所有快捷键"""
        for hotkey in list(self.registered_shortcuts.keys()):
            self.unregister(hotkey)
    
    def get_registered_shortcuts(self) -> Dict[str, str]:
        """获取已注册的快捷键列表"""
        return {k: v['description'] for k, v in self.registered_shortcuts.items()}
    
    def __del__(self):
```

```python
# core/config_watcher.py
import os
import threading
import time
from typing import Callable

class ConfigWatcher:
    def __init__(self, config_manager):
        self.config_manager = config_manager
        self.watch_files = []
        self.callbacks: Callable = []
        self.running = False
        self.thread: threading.Thread = None
    
    def add_watch(self, filepath: str):
        self.watch_files.append(filepath)
    
    def on_config_change(self, callback: Callable):
        self.callbacks.append(callback)
    
    def start(self):
        self.running = True
        self.thread = threading.Thread(target=self._watch_loop)
        self.thread.daemon = True
        self.thread.start()
    
    def stop(self):
        self.running = False
        if self.thread:
            self.thread.join(timeout=2)
    
    def _watch_loop(self):
        last_mtimes = {f: os.path.getmtime(f) for f in self.watch_files if os.path.exists(f)}
        
        while self.running:
            for filepath in self.watch_files:
                if not os.path.exists(filepath):
                    continue
                
                current_mtime = os.path.getmtime(filepath)
                if filepath in last_mtimes and current_mtime != last_mtimes[filepath]:
                    print(f"配置文件变更检测: {filepath}")
                    self.config_manager.load_all_configs()
                    for callback in self.callbacks:
                        callback(filepath)
                
                last_mtimes[filepath] = current_mtime
            
            time.sleep(1)
```
